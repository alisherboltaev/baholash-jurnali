rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // O'qituvchi: faqat shu Gmail bilan kirgan odam baho qo'ya oladi.
    // "SIZNING_GMAIL@gmail.com" ni o'zingizning Gmail manzilingizga almashtiring.
    function isTeacher() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email == "SIZNING_GMAIL@gmail.com";
    }

    // Baholar: hamma ko'radi (akkauntsiz), faqat o'qituvchi o'zgartiradi.
    match /groups/{groupId} {
      allow read: if true;
      allow write: if isTeacher();
    }

    // O'qituvchini tekshirish uchun yordamchi yozuv.
    match /meta/{docId} {
      allow read, write: if isTeacher();
    }
  }
}
