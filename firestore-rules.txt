rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Teacher: only the account signed in with this Gmail can edit grades.
    // Replace "YOUR_GMAIL@gmail.com" with your own Gmail address.
    function isTeacher() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email == "YOUR_GMAIL@gmail.com";
    }

    // Grades: anyone can read (no account needed), only the teacher can write.
    match /groups/{groupId} {
      allow read: if true;
      allow write: if isTeacher();
    }

    // Helper document used to verify the teacher.
    match /meta/{docId} {
      allow read, write: if isTeacher();
    }
  }
}
